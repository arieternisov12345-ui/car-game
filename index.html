Экран настройки игроков class SetupScreen extends StatefulWidget { @override _SetupScreenState createState() => _SetupScreenState(); } class _SetupScreenState extends State { final TextEditingController _nameController = TextEditingController(); final List _players = []; int _spyCount = 1; void _addPlayer() { if (_nameController.text.trim().isNotEmpty) { setState(() { _players.add(_nameController.text.trim()); _nameController.clear(); }); } } void _startGame() { if (_players.length < 3) { ScaffoldMessenger.of(context).showSnackBar( SnackBar(content: Text('Нужно минимум 3 игрока!')), ); return; } Navigator.push( context, MaterialPageRoute( builder: (context) => GameScreen(players: _players, spyCount: _spyCount), ), ); } @override Widget build(BuildContext context) { return Scaffold( appBar: AppBar(title: Text('Шпион — Создание игры')), body: Padding( padding: EdgeInsets.all(16.0), child: Column( children: [ TextField( controller: _nameController, decoration: InputDecoration( labelText: 'Имя игрока', suffixIcon: IconButton( icon: Icon(Icons.add), onPressed: _addPlayer, ), ), onSubmitted: (_) => _addPlayer(), ), SizedBox(height: 10), Expanded( child: ListView.builder( itemCount: _players.length, itemBuilder: (context, index) { return ListTile( title: Text('${index + 1}. ${_players[index]}'), trailing: IconButton( icon: Icon(Icons.delete, color: Colors.red), onPressed: () { setState(() { _players.removeAt(index); }); }, ), ); }, ), ), ElevatedButton( onPressed: _startGame, child: Text('Начать игру'), style: ElevatedButton.styleFrom(minimumSize: Size(double.infinity, 50)), ), ], ), ), ); } } // 2. Экран передачи телефона и просмотра ролей class GameScreen extends StatefulWidget { final List players; final int spyCount; GameScreen({required this.players, required this.spyCount}); @override _GameScreenState createState() => _GameScreenState(); } class _GameScreenState extends State { int currentIndex = 0; bool showRole = false; late List roles; final List locations = ['Банк', 'Пляж', 'Школа', 'Больница', 'Ресторан', 'Космолет']; @override void initState() { super.initState(); _generateRoles(); } void _generateRoles() { String secretLocation = locations[Random().nextInt(locations.length)]; roles = List.filled(widget.players.length, secretLocation); // Случайным образом назначаем шпионов var random = Random(); int assignedSpies = 0; while (assignedSpies < widget.spyCount) { int index = random.nextInt(widget.players.length); if (roles[index] != 'Шпион') { roles[index] = 'Шпион'; assignedSpies++; } } } void _nextPlayer() { setState(() { showRole = false; currentIndex++; }); } @override Widget build(BuildContext context) { if (currentIndex >= widget.players.length) { return Scaffold( appBar: AppBar(title: Text('Игра началась!')), body: Center( child: Column( mainAxisAlignment: MainAxisAlignment.center, children: [ Text('Все роли разданы!', style: TextStyle(fontSize: 24)), SizedBox(height: 20), ElevatedButton( onPressed: () => Navigator.pop(context), child: Text('В начало'), ), ], ), ), ); } return Scaffold( appBar: AppBar(title: Text('Игрок: ${widget.players[currentIndex]}')), body: Center( child: Padding( padding: EdgeInsets.all(24.0), child: Column( mainAxisAlignment: MainAxisAlignment.center, children: [ if (!showRole) ...[ Text( 'Передайте телефон игроку:\n\n${widget.players[currentIndex]}', textAlign: TextAlign.center, style: TextStyle(fontSize: 22), ), SizedBox(height: 30), ElevatedButton( onPressed: () => setState(() => showRole = true), child: Text('Посмотреть роль'), style: ElevatedButton.styleFrom(minimumSize: Size(200, 50)), ), ] else ...[ Text( 'Ваша роль:', style: TextStyle(fontSize: 18, color: Colors.grey), ), SizedBox(height: 10), Text( roles[currentIndex], textAlign: TextAlign.center, style: TextStyle( fontSize: 32, fontWeight: FontWeight.bold, color: roles[currentIndex] == 'Шпион' ? Colors.red : Colors.green, ), ), SizedBox(height: 40), ElevatedButton( onPressed: _nextPlayer, child: Text('Скрыть и передать дальше'), style: ElevatedButton.styleFrom(minimumSize: Size(200, 50)), ), ], ], ), ), ), ); } }

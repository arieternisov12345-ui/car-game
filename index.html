const playerCars = new Map(); mp.events.addCommand("car", (player, fullText) => { // Мисал: /car sultan const model = fullText && fullText.trim() ? fullText.trim() : "sultan"; // Эски машинаны өчүрүү const oldCar = playerCars.get(player.id); if (oldCar && mp.vehicles.exists(oldCar)) { oldCar.destroy(); } // Оюнчунун алдына чыгаруу const pos = player.position; const vehicle = mp.vehicles.new( mp.joaat(model), new mp.Vector3( pos.x + 3, pos.y, pos.z ), { heading: player.heading, numberPlate: "RPRP" } ); playerCars.set(player.id, vehicle); // Машинага отургузуу player.putIntoVehicle(vehicle, 0); player.outputChatBox( `!{00FF00}Машина чыгарылды: ${model}` ); }); // Машинаны өчүрүү mp.events.addCommand("delcar", (player) => { const vehicle = playerCars.get(player.id); if (!vehicle || !mp.vehicles.exists(vehicle)) { player.outputChatBox( "!{FF0000}Сенде машина жок." ); return; } vehicle.destroy(); playerCars.delete(player.id); player.outputChatBox( "!{00FF00}Машина өчүрүлдү." ); }); // Оюнчу чыкканда машинаны өчүрүү mp.events.add("playerQuit", (player) => { const vehicle = playerCars.get(player.id); if (vehicle && mp.vehicles.exists(vehicle)) { vehicle.destroy(); } playerCars.delete(player.id); });
